/* ============================================================ SAVE ORDER — FIXED ہر موجودہ Content Item کا order الگ محفوظ کیا جاتا ہے۔ اس سے "Order does not match content count" ختم ہوگا۔ ============================================================ */ async function saveOrder(){ const status = document.getElementById( "orderStatus" ); const items = Array.from( document.querySelectorAll( "#contentList .item" ) ); if(!items.length){ status.textContent = "No content to reorder."; return; } status.textContent = "Saving homepage order..."; try{ await getSession(); /* * موجودہ screen order حاصل کریں۔ */ const orderedItems = items.map( (item,index)=>({ id: Number( item.dataset.id ), order: index + 1 }) ); /* * Invalid IDs روکیں۔ */ const invalid = orderedItems.some( item => !Number.isFinite(item.id) || item.id <= 0 ); if(invalid){ throw new Error( "ایک یا زیادہ Content Items کا ID درست نہیں ہے۔" ); } /* * Backend کے موجودہ content record کو * ایک ایک کرکے update کریں۔ * * اس طریقے میں backend سے صرف order بدلتا ہے۔ * باقی content محفوظ رہتا ہے۔ */ for( const ordered of orderedItems ){ const current = await api( "/api/admin/content" ); const currentItems = Array.isArray( current.items ) ? current.items : []; const existing = currentItems.find( x => Number(x.id) === Number(ordered.id) ); if(!existing){ throw new Error( "Content #"+ ordered.id+ " موجود نہیں ہے۔" ); } await api( "/api/admin/content/update", { method:"POST", body: JSON.stringify({ id: Number( existing.id ), type: existing.type || "message", message: existing.message || "", button_text: existing.button_text || "", button_url: existing.button_url || "", media_url: existing.media_url || "", enabled: existing.enabled !== false, order: ordered.order }) } ); } status.textContent = "✓ Homepage order saved successfully."; /* * Backend سے دوبارہ تازہ ترتیب لائیں۔ */ await loadContent(); }catch(e){ status.textContent = "Order save failed: "+ e.message; } }
